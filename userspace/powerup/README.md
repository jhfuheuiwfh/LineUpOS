# scripts/build-native.js
// Compila powerup.js para binário nativo V8
// Elimina JIT, reduz memória drasticamente

import { build } from 'esbuild';
import { readFileSync, writeFileSync, mkdirSync, existsSync } from 'fs';
import { join } from 'path';

const DIST_DIR = join(__dirname, '..', 'dist');
const SNAPSHOT_DIR = join(__dirname, '..', 'snapshot');

async function buildNative() {
    console.log('[powerup.js] Building native V8 snapshot...');
    
    if (!existsSync(DIST_DIR)) mkdirSync(DIST_DIR, { recursive: true });
    if (!existsSync(SNAPSHOT_DIR)) mkdirSync(SNAPSHOT_DIR, { recursive: true });

    // 1. Bundle tudo (Next.js + React + nosso runtime) em um arquivo
    await build({
        entryPoints: [join(__dirname, '..', 'src', 'runtime', 'entry.ts')],
        bundle: true,
        platform: 'node',
        target: 'node20',
        format: 'cjs',
        outfile: join(DIST_DIR, 'bundle.js'),
        minify: true,
        treeShaking: true,
        external: ['v8', 'fs', 'path', 'os', 'crypto', 'buffer'],
        define: {
            'process.env.NODE_ENV': '"production"',
            'POWERUP_NATIVE': 'true',
            'POWERUP_AI_MEMORY': '732', // MB
            'globalThis.PowerUp': 'globalThis.PowerUp'
        },
        banner: {
            js: '// powerup.js native bundle - LineUpOS\n// DO NOT EDIT - Generated by build-native.js'
        }
    });

    console.log('[powerup.js] Bundle created:', join(DIST_DIR, 'bundle.js'));

    // 2. Gerar V8 startup snapshot
    // Requer Node.js compilado com --with-v8-snapshot
    // Veja: https://nodejs.org/api/cli.html#--snapshot-blob
    console.log('[powerup.js] Generating V8 snapshot...');
    console.log('[powerup.js] NOTE: Requires Node.js built with --with-v8-snapshot');
    console.log('[powerup.js] Run: ./configure --with-v8-snapshot && make -j$(nproc)');
    
    // Snapshot script (roda no Node com snapshot support)
    const snapshotScript = `
const v8 = require('v8');
const fs = require('fs');
const path = require('path');

const bundle = require('./dist/bundle.js');

// Inicializa runtime
const runtime = bundle.PowerUp.createApp({
    width: 1920,
    height: 1080,
    aiMemoryBudget: 732 * 1024 * 1024,
    ipcChannel: '/run/lineupos/powerup.sock'
});

// Serializa heap para snapshot
const snapshot = v8.serialize(runtime);
fs.writeFileSync(path.join('${SNAPSHOT_DIR}', 'powerup.blob'), snapshot);
console.log('Snapshot written to ${SNAPSHOT_DIR}/powerup.blob');
`;

    writeFileSync(join(SNAPSHOT_DIR, 'generate-snapshot.js'), snapshotScript);
    
    // 3. Build script para compilar Node com snapshot
    const buildNodeScript = `#!/bin/bash
# Build Node.js com V8 snapshot support
cd /tmp
git clone --depth 1 --branch v20.15.0 https://github.com/nodejs/node.git
cd node
./configure --with-v8-snapshot --without-npm --without-node-snapshot
make -j$(nproc)
cp out/Release/node ${SNAPSHOT_DIR}/node-snapshot
`;

    writeFileSync(join(SNAPSHOT_DIR, 'build-node-snapshot.sh'), buildNodeScript);
    
    console.log('[powerup.js] Build complete. Next steps:');
    console.log('  1. cd snapshot && chmod +x build-node-snapshot.sh && ./build-node-snapshot.sh');
    console.log('  2. ./node-snapshot generate-snapshot.js');
    console.log('  3. Copy powerup.blob to initramfs/usr/lib/powerup/');
}

buildNative().catch(err => {
    console.error('[powerup.js] Build failed:', err);
    process.exit(1);
});

// Scheduler cooperativo ultra-leve
// Substitui scheduler do React/Next.js

export type Priority = 'immediate' | 'high' | 'normal' | 'low' | 'idle';

export interface Task {
    id: number;
    priority: Priority;
    callback: () => void | Promise<void>;
    deadline?: number;
}

export class Scheduler {
    private queues: Map<Priority, Task[]> = new Map();
    private currentTask: Task | null = null;
    private taskId = 0;
    private running = false;
    
    constructor() {
        for (const p of ['immediate', 'high', 'normal', 'low', 'idle'] as Priority[]) {
            this.queues.set(p, []);
        }
    }
    
    schedule(callback: () => void | Promise<void>, priority: Priority = 'normal'): number {
        const task: Task = {
            id: ++this.taskId,
            priority,
            callback
        };
        this.queues.get(priority)!.push(task);
        this.tryFlush();
        return task.id;
    }
    
    cancel(taskId: number): boolean {
        for (const [, queue] of this.queues) {
            const idx = queue.findIndex(t => t.id === taskId);
            if (idx >= 0) {
                queue.splice(idx, 1);
                return true;
            }
        }
        return false;
    }
    
    private tryFlush() {
        if (this.running) return;
        this.running = true;
        
        // Microtask flush
        queueMicrotask(() => this.flush());
    }
    
    private flush() {
        this.running = false;
        
        while (true) {
            const task = this.popHighestPriority();
            if (!task) break;
            
            this.currentTask = task;
            try {
                const result = task.callback();
                if (result instanceof Promise) {
                    result.catch(e => console.error('Task error:', e));
                }
            } finally {
                this.currentTask = null;
            }
        }
    }
    
    private popHighestPriority(): Task | null {
        for (const p of ['immediate', 'high', 'normal', 'low', 'idle'] as Priority[]) {
            const queue = this.queues.get(p)!;
            if (queue.length > 0) return queue.shift()!;
        }
        return null;
    }
    
    getCurrentPriority(): Priority {
        return this.currentTask?.priority || 'normal';
    }
}

// Global scheduler
export const scheduler = new Scheduler();
